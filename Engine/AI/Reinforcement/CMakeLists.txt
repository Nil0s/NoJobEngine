cmake_minimum_required(VERSION 3.25)
project(NoJobRLValidation LANGUAGES CXX)
enable_testing()
get_filename_component(NOJOB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(test IN ITEMS ReinforcementTests TrackingLearningProof)
    add_executable(${test} ${test}.cpp ../Neural/NeuralNetwork.cpp)
    target_include_directories(${test} PRIVATE "${NOJOB_ROOT}")
    target_compile_features(${test} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
