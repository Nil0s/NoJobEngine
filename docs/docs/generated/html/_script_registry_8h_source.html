40 template<typename TScript, typename T>
41 ScriptFieldDefinition MakeMemberFieldDefinition(
42 const char* name, T TScript::* member, T defaultValue)
43 {
44 ScriptFieldDefinition field;
45 field.Name = name;
46 field.DefaultValue = MakeScriptFieldValue(defaultValue);
47 field.SetMember = [member](Script& script, const ScriptFieldValue& value)
48 {
49 if (auto* typed = dynamic_cast<TScript*>(&script))
50 typed->*member = ReadScriptFieldValue<T>(value);
51 };
52 field.GetMember = [member](const Script& script)
53 {
54 if (const auto* typed = dynamic_cast<const TScript*>(&script))
55 return MakeScriptFieldValue(typed->*member);
56 return ScriptFieldValue{};
57 };
58 return field;
59 }

128 static void ApplyDefaults(NativeScriptComponent& component)
129 {
130 const auto* def = Find(component.ScriptName);
131 if (!def) return;
132
133 // V1.9 migration: numeric TreeSlot -> named .njbt asset.
134 if (component.ScriptName == "BTAssetAgent" &&
135 !component.Fields.contains("BehaviorTreeAsset"))
136 {
137 const auto old = component.Fields.find("TreeSlot");
138 if (old != component.Fields.end() && old->second.Type == ScriptFieldType::Int)
139 {
140 const int slot = old->second.Int;
141 component.Fields["BehaviorTreeAsset"] = ScriptFieldValue::MakeString(
142 slot == 0 ? "Default" : "Tree_" + std::to_string(slot));
143 }
144 }
145 std::unordered_map<std::string, ScriptFieldValue> migrated;
146 for (const auto& field : def->Fields)
147 {
148 const auto existing = component.Fields.find(field.Name);
149 if (existing != component.Fields.end() &&
150 existing->second.Type == field.DefaultValue.Type)
151 migrated[field.Name] = existing->second;
152 else
153 migrated[field.Name] = field.DefaultValue;
154 }
155 component.Fields = std::move(migrated);
156 }
