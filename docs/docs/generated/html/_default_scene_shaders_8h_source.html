110 vec2 st1=dFdx(v_TexCoord), st2=dFdy(v_TexCoord);

112 // Reconstruct the full cotangent frame from position/UV
113 // derivatives. The previous B=-cross(N,T) assumed one fixed
114 // UV handedness, which breaks on mirrored FBX/Mixamo UV
115 // islands and can turn correctly textured areas almost black.
116 float det=st1.x*st2.y-st1.y*st2.x;
117 if(abs(det)<0.000001) return N;
118 vec3 T=(Q1*st2.y-Q2*st1.y)/det;
119 vec3 B=(-Q1*st2.x+Q2*st1.x)/det;
