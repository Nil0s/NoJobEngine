90 // Native patrol task: four waypoints around the agent's spawn point.
91 // Runtime progress is stored per agent in the Blackboard.
92 r->RegisterAction("Patrol", [](BTContext& ctx, float dt) {
93 auto* script = BTGetScript(ctx);
94 if (!script || !script->HasComponent<NavAgentComponent>() ||
95 !script->HasComponent<TransformComponent>()) return BTStatus::Failure;
96 auto* scene = script->GetEntity().GetScene();
97 if (!scene) return BTStatus::Failure;
98 auto& nav = script->GetComponent<NavAgentComponent>();
99 if (!nav.Enabled) return BTStatus::Failure;
100 const glm::vec3 position = glm::vec3(scene->GetWorldTransform(script->GetEntity())[3]);
101 auto& bb = ctx.Blackboard();
102 if (!bb.Contains("PatrolOrigin")) bb.Set("PatrolOrigin", position);
103 const glm::vec3 origin = bb.Get<glm::vec3>("PatrolOrigin");
104 const float radius = std::max(bb.GetOr<float>("PatrolRadius", 3.f), 0.25f);
105 const float pause = std::max(bb.GetOr<float>("PatrolWaitSeconds", 0.4f), 0.f);
106 int index = bb.GetOr<int>("PatrolIndex", 0) % 4;
107 if (index < 0) index += 4;
108 const glm::vec3 offsets[4] = {
109 {radius,0.f,0.f}, {radius,0.f,radius}, {0.f,0.f,radius}, {0.f,0.f,0.f}
110 };
111 const glm::vec3 destination = origin + offsets[index];
112 glm::vec3 difference = destination - position;
113 difference.y = 0.f;
114 const float stoppingDistance = std::max(nav.StoppingDistance, 0.15f);
115 if (glm::dot(difference,difference) <= stoppingDistance*stoppingDistance) {
116 nav.HasDestination = false;
117 const float elapsed = bb.GetOr<float>("PatrolWaitElapsed", 0.f) + std::max(dt,0.f);
118 if (elapsed >= pause) {
119 bb.Set("PatrolIndex", (index+1)%4);
120 bb.Set("PatrolWaitElapsed", 0.f);
121 } else bb.Set("PatrolWaitElapsed", elapsed);
122 } else {
123 bb.Set("PatrolWaitElapsed", 0.f);
124 nav.Destination = destination;
125 nav.HasDestination = true;
126 }
127 ctx.SetCancellation(ctx.ActiveNode(), [](BTContext& c) {
128 if (auto* owner = BTGetScript(c); owner && owner->HasComponent<NavAgentComponent>())
129 owner->GetComponent<NavAgentComponent>().HasDestination = false;
130 });
131 return BTStatus::Running;
132 });
133 r->RegisterAction("StopMoving", [](BTContext& ctx,float) {
134 auto* s=BTGetScript(ctx);
135 if (!s || !s->HasComponent<NavAgentComponent>()) return BTStatus::Failure;
136 s->GetComponent<NavAgentComponent>().HasDestination=false;
137 return BTStatus::Success;
138 });
139 r->RegisterAction("MoveForward", [](BTContext& ctx,float dt) {
140 auto* s=BTGetScript(ctx);
141 if (!s || !s->HasComponent<TransformComponent>()) return BTStatus::Failure;
142 s->GetComponent<TransformComponent>().Position.x += dt;
143 return BTStatus::Running;
144 });
145 return r;
146}
