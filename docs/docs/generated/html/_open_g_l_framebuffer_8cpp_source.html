30 GLuint CreatePostProgram()
31 {
32 const char* vs = R"(
33 #version 460 core
34 const vec2 P[3]=vec2[3](vec2(-1.0,-1.0),vec2(3.0,-1.0),vec2(-1.0,3.0));
35 out vec2 v_UV;
36 void main(){v_UV=P[gl_VertexID]*0.5+0.5;gl_Position=vec4(P[gl_VertexID],0.0,1.0);}
37 )";
38 const char* fs = R"(
39 #version 460 core
40 in vec2 v_UV; layout(location=0) out vec4 o_Color;
41 uniform sampler2D u_HDR;
42 uniform sampler2D u_Depth;
43 uniform vec2 u_InvResolution;
44 uniform float u_Exposure,u_BloomThreshold,u_BloomStrength,u_AOIntensity;
45 uniform int u_Bloom,u_FXAA,u_AO;
46
47 vec3 SampleHDR(vec2 uv){return texture(u_HDR,clamp(uv,vec2(0.0),vec2(1.0))).rgb;}
48 float Luma(vec3 c){return dot(c,vec3(0.299,0.587,0.114));}
49 vec3 ACES(vec3 x){
50 const float a=2.51,b=0.03,c=2.43,d=0.59,e=0.14;
51 return clamp((x*(a*x+b))/(x*(c*x+d)+e),0.0,1.0);
52 }
53 vec3 FXAA(vec2 uv){
54 vec3 rgbM=SampleHDR(uv);
55 float lumaM=Luma(rgbM);
56 float lumaN=Luma(SampleHDR(uv+vec2(0.0,-u_InvResolution.y)));
57 float lumaS=Luma(SampleHDR(uv+vec2(0.0, u_InvResolution.y)));
58 float lumaW=Luma(SampleHDR(uv+vec2(-u_InvResolution.x,0.0)));
59 float lumaE=Luma(SampleHDR(uv+vec2( u_InvResolution.x,0.0)));
60 float mn=min(lumaM,min(min(lumaN,lumaS),min(lumaW,lumaE)));
61 float mx=max(lumaM,max(max(lumaN,lumaS),max(lumaW,lumaE)));
62 if(mx-mn < max(0.0312,mx*0.125)) return rgbM;
63 vec2 dir=vec2(-(lumaN-lumaS),lumaW-lumaE);
64 dir=clamp(dir/(min(abs(dir.x),abs(dir.y))+0.0001),vec2(-8.0),vec2(8.0))*u_InvResolution;
65 return (SampleHDR(uv+dir*(1.0/3.0-0.5))+SampleHDR(uv+dir*(2.0/3.0-0.5)))*0.5;
66 }
67 float ScreenAO(vec2 uv){
68 float center=texture(u_Depth,uv).r;
69 if(center>=0.99999) return 1.0;
70 float occ=0.0;
71 const vec2 dirs[8]=vec2[8](vec2(1,0),vec2(-1,0),vec2(0,1),vec2(0,-1),
72 vec2(1,1),vec2(-1,1),vec2(1,-1),vec2(-1,-1));
73 for(int i=0;i<8;i++){
74 float d=texture(u_Depth,uv+dirs[i]*u_InvResolution*3.0).r;
75 occ += smoothstep(0.0004,0.012,center-d);
76 }
77 return 1.0-(occ/8.0)*u_AOIntensity;
78 }
79 vec3 Bloom(vec2 uv){
80 vec3 b=vec3(0.0); float w=0.0;
81 for(int x=-4;x<=4;x++) for(int y=-4;y<=4;y++){
82 vec2 o=vec2(x,y)*u_InvResolution*2.0;
83 vec3 c=SampleHDR(uv+o);
84 float bright=max(Luma(c)-u_BloomThreshold,0.0);
85 float ww=exp(-float(x*x+y*y)/10.0);
86 b+=c*(bright/(bright+1.0))*ww; w+=ww;
87 }
88 return b/max(w,0.0001);
89 }
90 void main(){
91 vec3 hdr=(u_FXAA==1)?FXAA(v_UV):SampleHDR(v_UV);
92 if(u_AO==1) hdr*=ScreenAO(v_UV);
93 if(u_Bloom==1) hdr+=Bloom(v_UV)*u_BloomStrength;
94 vec3 mapped=ACES(hdr*max(u_Exposure,0.001));
95 mapped=pow(mapped,vec3(1.0/2.2));
96 o_Color=vec4(mapped,1.0);
97 }
98 )";
99 GLuint v=CompileStage(GL_VERTEX_SHADER,vs), f=CompileStage(GL_FRAGMENT_SHADER,fs);
100 GLuint program=glCreateProgram();
101 glAttachShader(program,v); glAttachShader(program,f); glLinkProgram(program);
102 glDeleteShader(v); glDeleteShader(f);
103 GLint ok=GL_FALSE; glGetProgramiv(program,GL_LINK_STATUS,&ok);
104 if(!ok){glDeleteProgram(program);throw std::runtime_error("NoJobEngine post-process shader link failed.");}
105 return program;
106 }

183 void OpenGLFramebuffer::RunPostProcess()
184 {
185 // Post-processing is executed in the middle of the editor render loop.
186 // Preserve every GL state we touch; ImGui's multi-viewport Win32 backend
187 // may switch platform windows/contexts later in glfwPollEvents().
188 GLint previousDrawFBO = 0;
189 GLint previousReadFBO = 0;
190 GLint previousProgram = 0;
191 GLint previousVAO = 0;
192 GLint previousViewport[4] = { 0, 0, 0, 0 };
193 const GLboolean depthWasEnabled = glIsEnabled(GL_DEPTH_TEST);
194 glGetIntegerv(GL_DRAW_FRAMEBUFFER_BINDING, &previousDrawFBO);
195 glGetIntegerv(GL_READ_FRAMEBUFFER_BINDING, &previousReadFBO);
196 glGetIntegerv(GL_CURRENT_PROGRAM, &previousProgram);
197 glGetIntegerv(GL_VERTEX_ARRAY_BINDING, &previousVAO);
198 glGetIntegerv(GL_VIEWPORT, previousViewport);
199
200 glBindFramebuffer(GL_FRAMEBUFFER,m_PostProcessFBO);
201 glViewport(0,0,(GLsizei)m_Specification.Width,(GLsizei)m_Specification.Height);
202 glDisable(GL_DEPTH_TEST);
203 glUseProgram(m_PostProgram);
204 glBindTextureUnit(0,m_ColorAttachment);
205 glBindTextureUnit(1,m_DepthAttachment);
206 glUniform1i(glGetUniformLocation(m_PostProgram,"u_HDR"),0);
207 glUniform1i(glGetUniformLocation(m_PostProgram,"u_Depth"),1);
208 glUniform2f(glGetUniformLocation(m_PostProgram,"u_InvResolution"),
209 1.0f/(float)m_Specification.Width,1.0f/(float)m_Specification.Height);
210 glUniform1f(glGetUniformLocation(m_PostProgram,"u_Exposure"),m_Specification.Exposure);
211 glUniform1f(glGetUniformLocation(m_PostProgram,"u_BloomThreshold"),m_Specification.BloomThreshold);
212 glUniform1f(glGetUniformLocation(m_PostProgram,"u_BloomStrength"),m_Specification.BloomStrength);
213 glUniform1f(glGetUniformLocation(m_PostProgram,"u_AOIntensity"),m_Specification.AOIntensity);
214 glUniform1i(glGetUniformLocation(m_PostProgram,"u_Bloom"),m_Specification.Bloom?1:0);
215 glUniform1i(glGetUniformLocation(m_PostProgram,"u_FXAA"),m_Specification.FXAA?1:0);
216 glUniform1i(glGetUniformLocation(m_PostProgram,"u_AO"),m_Specification.ScreenSpaceAO?1:0);
217 glBindVertexArray(m_FullscreenVAO);
218 glDrawArrays(GL_TRIANGLES,0,3);
219 glBindVertexArray(static_cast<GLuint>(previousVAO));
220 glUseProgram(static_cast<GLuint>(previousProgram));
221 if(depthWasEnabled) glEnable(GL_DEPTH_TEST); else glDisable(GL_DEPTH_TEST);
222 glBindFramebuffer(GL_DRAW_FRAMEBUFFER,static_cast<GLuint>(previousDrawFBO));
223 glBindFramebuffer(GL_READ_FRAMEBUFFER,static_cast<GLuint>(previousReadFBO));
224 glViewport(previousViewport[0],previousViewport[1],previousViewport[2],previousViewport[3]);
225 }

248 void OpenGLFramebuffer::Resize(std::uint32_t width,std::uint32_t height)
249 {
250 // Docking/minimize can briefly report invalid sizes. Never recreate GL
251 // resources from those transient values.
252 if(width==0||height==0||width>16384||height>16384) return;
