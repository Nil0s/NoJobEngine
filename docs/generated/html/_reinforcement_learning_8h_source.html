50class ReplayBuffer {
51public:
52 explicit ReplayBuffer(size_t capacity) : capacity_(capacity) {
53 if (!capacity) throw std::invalid_argument("Replay capacity must be positive");
54 }
55 void Push(Transition transition) {
56 if (data_.size() == capacity_) data_.pop_front();
57 data_.push_back(std::move(transition));
58 }
59 size_t Size() const { return data_.size(); }
60 std::vector<Transition> Sample(size_t count, std::mt19937& rng) const {
61 if (count > data_.size()) throw std::invalid_argument("Not enough replay samples");
62 std::vector<size_t> ids(data_.size());
63 for (size_t i=0;i<ids.size();++i) ids[i]=i;
64 std::shuffle(ids.begin(),ids.end(),rng);
65 std::vector<Transition> batch; batch.reserve(count);
66 for (size_t i=0;i<count;++i) batch.push_back(data_[ids[i]]);
67 return batch;
68 }
69private:
70 size_t capacity_;
71 std::deque<Transition> data_;
72};

97 DQNAgent(size_t observationSize, int actions, Options options = {})
98 : observations_(observationSize), actions_(actions), options_(options), rng_(options.seed),
99 replay_(options.replayCapacity), online_(options.seed), target_(options.seed), epsilon_(options.epsilonStart) {
100 if (!observations_ || actions_ < 2 || actions_ > 128 || !options_.batchSize ||
101 !options_.targetSyncSteps || !options_.hiddenUnits || !std::isfinite(options_.gamma) ||
102 options_.gamma < 0 || options_.gamma > 1 || !std::isfinite(options_.learningRate) ||
103 options_.learningRate <= 0 || options_.epsilonEnd < 0 || options_.epsilonEnd > 1 ||
104 options_.epsilonStart < options_.epsilonEnd || options_.epsilonStart > 1 ||
105 options_.epsilonDecay <= 0 || options_.epsilonDecay > 1)
106 throw std::invalid_argument("Invalid DQN configuration");
107 online_.AddDense(observations_, options_.hiddenUnits, Neural::Activation::ReLU);
108 online_.AddDense(options_.hiddenUnits, static_cast<size_t>(actions_), Neural::Activation::Linear);
109 target_ = online_;
110 }

112 int Act(const std::vector<float>& observation, bool greedy = false) {
113 Validate(observation);
114 if (!greedy && std::uniform_real_distribution<float>(0.f,1.f)(rng_) < epsilon_)
115 return std::uniform_int_distribution<int>(0,actions_-1)(rng_);
116 const auto q = online_.Predict(observation);
117 return static_cast<int>(std::distance(q.begin(),std::max_element(q.begin(),q.end())));
118 }
119
120 bool Observe(Transition transition) {
121 Validate(transition.observation); Validate(transition.nextObservation);
122 if (transition.action < 0 || transition.action >= actions_ || !std::isfinite(transition.reward))
123 throw std::invalid_argument("Invalid transition");
124 replay_.Push(std::move(transition));
125 if (replay_.Size() < std::max(options_.warmup,options_.batchSize)) return false;
126 const auto batch = replay_.Sample(options_.batchSize,rng_);
127 std::vector<std::vector<float>> xs,ys;
128 xs.reserve(batch.size()); ys.reserve(batch.size());
129 for (const auto& sample:batch) {
130 auto q = online_.Predict(sample.observation);
131 float bellman = sample.reward;
132 if (!sample.terminal) {
133 const auto nextOnline = online_.Predict(sample.nextObservation);
134 const auto best = static_cast<size_t>(std::distance(nextOnline.begin(),
135 std::max_element(nextOnline.begin(),nextOnline.end())));
136 bellman += options_.gamma * target_.Predict(sample.nextObservation)[best];
137 }
138 q[static_cast<size_t>(sample.action)] = bellman;
139 xs.push_back(sample.observation); ys.push_back(std::move(q));
140 }
141 Neural::TrainingOptions training;
142 training.epochs=1; training.batchSize=options_.batchSize;
143 training.learningRate=options_.learningRate; training.adam=true;
144 training.seed=options_.seed + static_cast<uint32_t>(updates_);
145 auto history=online_.Train(xs,ys,training);
146 lastLoss_=history.epochLoss.back();
147 if (++updates_ % options_.targetSyncSteps == 0) target_=online_;
148 return true;
149 }

151 void EndEpisode() { epsilon_=std::max(options_.epsilonEnd,epsilon_*options_.epsilonDecay); }

180class TrackingEnvironment final : public Environment {
181public:
182 size_t ObservationSize() const override { return 2; }
183 int ActionCount() const override { return 3; }
184 std::vector<float> Reset(uint32_t seed) override {
185 std::mt19937 rng(seed);
186 position_=std::uniform_real_distribution<float>(-.9f,.9f)(rng);
187 target_=std::uniform_real_distribution<float>(-.9f,.9f)(rng);
188 ticks_=0; return {position_,target_-position_};
189 }
190 StepResult Step(int action) override {
191 if (action<0 || action>=ActionCount()) throw std::invalid_argument("Invalid tracking action");
192 position_=std::clamp(position_ + static_cast<float>(action-1)*.12f,-1.f,1.f);
193 const float distance=std::abs(target_-position_);
194 ++ticks_;
195 return {{position_,target_-position_},distance<.08f?1.f:-distance*.1f,
196 distance<.08f || ticks_>=40};
197 }
198private:
199 float position_=0.f,target_=0.f; size_t ticks_=0;

202template<class Callback>
203void Train(Environment& environment, DQNAgent& agent, size_t episodes,
204 size_t maxSteps, uint32_t seed, Callback callback) {
205 if (!maxSteps) throw std::invalid_argument("Episode step limit must be positive");
206 for (size_t episode=0;episode<episodes;++episode) {
207 auto observation=environment.Reset(seed+static_cast<uint32_t>(episode));
208 EpisodeMetrics metrics;
209 for (size_t step=0;step<maxSteps;++step) {
210 const int action=agent.Act(observation);
211 auto result=environment.Step(action);
212 const bool terminal=result.terminated || step+1==maxSteps;
213 agent.Observe({observation,action,result.reward,result.observation,terminal});
214 metrics.totalReward+=result.reward; metrics.steps=step+1;
215 observation=std::move(result.observation);
216 if (terminal) break;
217 }
218 agent.EndEpisode(); metrics.epsilon=agent.Epsilon(); metrics.loss=agent.LastLoss();
219 if (!callback(episode+1,metrics)) break;
220 }
221}
