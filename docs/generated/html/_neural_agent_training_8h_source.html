34inline std::vector<std::vector<float>> Activations(const Neural::NeuralNetwork& net, const std::vector<float>& input) {
35 std::vector<std::vector<float>> result{input}; auto values=input;
36 for(const auto& layer:net.Layers()) {
37 if(values.size()!=layer.Inputs()) throw std::runtime_error("Neural input dimensions mismatch");
38 std::vector<float> next(layer.Outputs());
39 for(size_t i=0;i<layer.Outputs();++i) {
40 float sum=layer.Biases()[i];
41 for(size_t j=0;j<values.size();++j) sum+=layer.Weights()(i,j)*values[j];
42 next[i]=Neural::Activate(sum,layer.Function());
43 }
44 result.push_back(next); values=std::move(next);
45 }
46 return result;
47}

66inline Neural::TrainingHistory TrainPreset(const std::string& outputName,
67 const std::string& sourceName, TrainingPreset preset,
68 int epochs, float learningRate, bool resume, bool overwrite) {
69 const auto output = NeuralAssetPath(outputName); // also validates name
70 if (resume && sourceName.empty()) throw std::runtime_error("Select a source model to resume training");
71 if (std::filesystem::exists(output) && !overwrite)
72 throw std::runtime_error("Output already exists. Choose another Save As name or enable Overwrite Output");
73 Neural::NeuralNetwork net(42);
74 const size_t nIn = preset == TrainingPreset::FollowXZ ? 2 : (preset == TrainingPreset::TrackY ? 3 : 6);
75 const size_t nOut = preset == TrainingPreset::FollowXZ ? 2 : 1;
76 if (resume) {
77 net = Neural::NeuralNetwork::Load(NeuralAssetPath(sourceName).string());
78 if (net.Layers().empty() || net.Layers().front().Inputs()!=nIn || net.Layers().back().Outputs()!=nOut)
79 throw std::runtime_error("Source model dimensions do not match the selected training preset");
80 } else if (preset == TrainingPreset::FollowXZ) {
81 net.AddDense(2,8,Neural::Activation::Tanh);
82 net.AddDense(8,2,Neural::Activation::Tanh);
83 } else if (preset == TrainingPreset::TrackY) {
84 net.AddDense(3,12,Neural::Activation::Tanh);
85 net.AddDense(12,1,Neural::Activation::Tanh);
86 } else {
87 net.AddDense(6,12,Neural::Activation::Tanh);
88 net.AddDense(12,8,Neural::Activation::Tanh);
89 net.AddDense(8,1,Neural::Activation::Tanh);
90 }
91 std::vector<std::vector<float>> xs, ys;
92 if (preset == TrainingPreset::FollowXZ) {
93 for (int x=-8;x<=8;++x) for (int z=-8;z<=8;++z) {
94 const float dx=x/8.f,dz=z/8.f,len=std::sqrt(dx*dx+dz*dz);
95 xs.push_back({dx,dz});
96 ys.push_back(len>1e-5f?std::vector<float>{dx/len,dz/len}:std::vector<float>{0.f,0.f});
97 }
98 } else if (preset == TrainingPreset::TrackY) {
99 // Inputs: normalized target Y, self Y, relative Y. Output: vertical direction.
100 for(int target=-10;target<=10;++target) for(int self=-10;self<=10;++self) {
101 float t=target/10.f, p=self/10.f, d=t-p;
102 xs.push_back({t,p,std::clamp(d,-1.f,1.f)});
103 ys.push_back({std::clamp(d*3.f,-1.f,1.f)});
104 }
105 } else {
106 // Inputs: ball X/Y, velocity X/Y, paddle Y, horizontal distance.
107 // Labels use a deterministic intercept heuristic, not reinforcement learning.
108 for(int i=0;i<6000;++i) {
109 const float bx=static_cast<float>((i*37)%199-99)/100.f;
110 const float by=static_cast<float>((i*53)%199-99)/100.f;
111 const float vx=static_cast<float>((i*17)%161-80)/100.f;
112 const float vy=static_cast<float>((i*29)%161-80)/100.f;
113 const float py=static_cast<float>((i*71)%199-99)/100.f;
114 const float dx=std::clamp((1.f-bx)*0.5f,0.f,1.f);
115 const float time=vx>0.05f ? std::min(3.f,(1.f-bx)/vx) : 0.f;
116 float intercept=by+vy*time;
117 // Reflect off normalized +/-1 playfield bounds.
118 while(intercept>1.f || intercept< -1.f) {
119 if(intercept>1.f) intercept=2.f-intercept;
120 if(intercept< -1.f) intercept=-2.f-intercept;
121 }
122 xs.push_back({bx,by,vx,vy,py,dx});
123 ys.push_back({std::clamp((intercept-py)*3.f,-1.f,1.f)});
124 }
125 }
126 Neural::TrainingOptions options;
127 options.epochs=static_cast<size_t>(std::clamp(epochs,1,20000));
128 options.learningRate=std::clamp(learningRate,0.00001f,0.1f);
129 options.batchSize=16; options.adam=true; options.seed=42;
130 auto history=net.Train(xs,ys,options);
131 std::filesystem::create_directories(output.parent_path());
132 net.Save(output.string());
133 return history;
134}
