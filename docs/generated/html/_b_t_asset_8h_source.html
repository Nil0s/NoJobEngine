40 void Validate() const {
41 if (Nodes.empty() || Nodes.size() > 4096) throw std::runtime_error("Invalid BT node count");
42 std::unordered_set<BTNodeID> ids;
43 std::size_t roots = 0;
44 for (const auto& n : Nodes) {
45 if (!n.ID || !ids.insert(n.ID).second) throw std::runtime_error("Duplicate/zero BT node ID");
46 if (n.Parent == 0) ++roots;
47 if (n.Kind != "Sequence" && n.Kind != "Selector" && n.Kind != "Action" && n.Kind != "Condition" &&
48 n.Kind != "Decorator" && n.Kind != "Service" && n.Kind != "Parallel")
49 throw std::runtime_error("Unknown BT node kind");
50 if ((n.Kind == "Action" || n.Kind == "Condition" || n.Kind == "Decorator" || n.Kind == "Service") && (n.Native.empty() || n.Native == "-"))

53 if (roots != 1) throw std::runtime_error("BT requires exactly one root");
54 for (const auto& n : Nodes) if (n.Parent && !ids.contains(n.Parent))
55 throw std::runtime_error("BT parent not found");
56 for (const auto& n : Nodes) {

58 auto current = n.ID;
59 while (current) {
60 if (!visited.insert(current).second) throw std::runtime_error("BT cycle detected");
61 const BTAssetNode* node = nullptr;
62 for (const auto& candidate : Nodes) if (candidate.ID == current) {node=&candidate;break;}
63 if (!node) throw std::runtime_error("BT missing ancestor");
64 current = node->Parent;
65 }
66 }
67 for (const auto& n : Nodes) {
68 std::size_t count=0;
69 for (const auto& child : Nodes) if (child.Parent==n.ID) ++count;
70 if ((n.Kind=="Decorator" || n.Kind=="Service") && count!=1)

72 if (n.Kind=="Parallel" && count!=2)

92 static BTAsset Load(const std::filesystem::path& path) {
93 std::ifstream in(path);
94 if (!in) throw std::runtime_error("Cannot read BT asset: " + path.string());
95 std::string magic; int version = 0;
96 if (!(in >> magic >> version) || magic != "NJOB_BT" || (version != 1 && version != 2))
97 throw std::runtime_error("Unsupported BT asset version");
98 BTAsset asset;
99 std::string tag;
100 while (in >> tag) {
101 if (tag != "NODE") throw std::runtime_error("Unexpected BT asset token");
102 BTAssetNode node;
103 if (!(in >> node.ID >> node.Parent >> node.Kind >> node.Native)) throw std::runtime_error("Truncated BT asset");
104 if (version == 2 && !(in >> node.Parameter >> node.Interval))
105 throw std::runtime_error("Truncated BT v2 asset");
106 asset.Nodes.push_back(std::move(node));
107 }
108 if (!in.eof()) throw std::runtime_error("Invalid BT asset syntax");
109 asset.Validate();
110 return asset;

117 for (const auto& n : Nodes) if (n.ID == id) { node = &n; break; }
118 if (!node) throw std::runtime_error("BT node missing");

124 node->Parameter == "-" ? "" : node->Parameter, make(child));
125 return std::make_unique<BTService>(id,registry,node->Native,node->Interval,make(child));
126 }
127 if (node->Kind == "Parallel") {
128 std::vector<BTNodeID> children;
129 for (const auto& n : Nodes) if (n.Parent == id) children.push_back(n.ID);
130 return std::make_unique<BTParallel>(id,make(children[0]),make(children[1]));
131 }
132 if (node->Kind == "Action") return std::make_unique<BTActionNode>(id,registry,node->Native);
133 if (node->Kind == "Condition") return std::make_unique<BTConditionNode>(id,registry,node->Native);
134 if (node->Kind == "Sequence") {
135 auto result = std::make_unique<BTSequence>(id);
136 for (const auto& n : Nodes) if (n.Parent == id) result->AddChild(make(n.ID));
137 return result;
138 }
139 auto result = std::make_unique<BTSelector>(id);
140 for (const auto& n : Nodes) if (n.Parent == id) result->AddChild(make(n.ID));
141 return result;
142 };
143 for (const auto& n : Nodes) if (!n.Parent) return std::make_shared<const BTTree>(make(n.ID));
