23 // This also supports ordinary Condition nodes, without a decorator

30 for (std::size_t i = 0; i < runningIndex; ++i) {
31 const auto status = m_Children[i]->Tick(context, dt);
32 if (status == BTStatus::Failure) {
33 m_Children[i]->Reset(context);
34 continue;
35 }
36 context.Emit(GetID(), "PriorityChanged", "Higher priority branch preempted running fallback");

38 context.GetNodeState(GetID()).CurrentChild = i;
39 if (status == BTStatus::Success) {
40 Reset(context);
41 return BTStatus::Success;
42 }
43 return BTStatus::Running;
44 }
45 }
46 while (true) {
47 const auto index = context.GetNodeState(GetID()).CurrentChild;
48 if (index >= m_Children.size()) {
49 Reset(context);
50 return BTStatus::Failure;

53 if (status == BTStatus::Running) return BTStatus::Running;
54 if (status == BTStatus::Success) {

57 }
