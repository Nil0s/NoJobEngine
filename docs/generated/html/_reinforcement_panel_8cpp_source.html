25void ReinforcementPanel::Start() {

27 Stop();

34 {
35 std::lock_guard<std::mutex> lock(mutex_);

37 }
38 cancel_=false; completed_=0; running_=true;
39 const int episodes=episodes_, maxSteps=maxSteps_, batch=batch_, hidden=hidden_;
40 const float rate=learningRate_, gamma=gamma_;
41 const bool obstacles=task_==1;
42 worker_=std::thread([this,path,episodes,maxSteps,batch,hidden,rate,gamma,obstacles] {
43 try {
44 RL::Options options=obstacles?RL::ObstacleOptions():RL::Options{};
45 options.batchSize=static_cast<size_t>(batch);
46 options.hiddenUnits=static_cast<size_t>(hidden);
47 options.learningRate=rate; options.gamma=gamma;
48 options.warmup=std::max<size_t>(static_cast<size_t>(batch),128);
49 std::unique_ptr<RL::Environment> env;
50 if(obstacles)env=std::make_unique<RL::ObstacleEnvironment>();else env=std::make_unique<RL::TrackingEnvironment>();
51 RL::DQNAgent agent(env->ObservationSize(),env->ActionCount(),options);
52 RL::Train(*env,agent,static_cast<size_t>(episodes),static_cast<size_t>(maxSteps),42,
53 [this](size_t episode,const RL::EpisodeMetrics& metrics) {
54 {
55 std::lock_guard<std::mutex> lock(mutex_);
56 rewards_.push_back(metrics.totalReward);
57 losses_.push_back(metrics.loss);
58 }
59 completed_=static_cast<int>(episode);
60 return !cancel_.load();
61 });
62 if(!cancel_) {
63 std::filesystem::create_directories(path.parent_path());
64 if(obstacles){
65 {std::lock_guard<std::mutex> lock(mutex_);status_="Refining stalled core cases (3000 reward-driven episodes)...";}
66 auto refined=RL::RefineObstaclePolicy(agent.Policy(),[this,episodes](size_t n,const RL::EpisodeMetrics&m){
67 {std::lock_guard<std::mutex> lock(mutex_);rewards_.push_back(m.totalReward);losses_.push_back(m.loss);}
68 completed_=episodes+static_cast<int>(n);return !cancel_.load();
69 });
70 if(refined&&!cancel_)refined->Save(path.string());
71 }else agent.SavePolicy(path.string());
72 }
73 std::lock_guard<std::mutex> lock(mutex_);
74 status_=cancel_?"Training cancelled":"Policy saved: "+path.string();
75 } catch(const std::exception& e) {
76 std::lock_guard<std::mutex> lock(mutex_); status_=e.what();
77 }
78 running_=false;
79 });

131 else{obstacleReport_=std::move(result);status_="Obstacle comparison complete: learned / untrained / random / native NavMesh.";}

133 RL::TrackingReport settings;
134 settings.policy=path.string();
135 auto result=RL::EvaluateTracking(model,settings,[this](size_t count) {
136 completed_=static_cast<int>(count); return !cancel_.load();
137 });
138 std::lock_guard<std::mutex> lock(mutex_);
139 if(!result.complete || cancel_) status_="Evaluation cancelled; no report published.";
140 else { report_=std::move(result); status_="Evaluation complete (saved policy; scene unchanged)."; }
141 }

143 std::lock_guard<std::mutex> lock(mutex_); status_=e.what();
