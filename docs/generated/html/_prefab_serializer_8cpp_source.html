281Entity PrefabSerializer::Revert(Entity instance,const std::shared_ptr<Mesh>& fallback,const std::shared_ptr<Material>& base)
282{

286 Entity parent=scene->GetParent(instance);
287
288 // Destroy the whole current instance hierarchy. Scene::DestroyEntity normally
289 // preserves children as roots, which is not what prefab Revert should do.
290 std::function<void(Entity)> destroyTree=[&](Entity e){
291 auto children=scene->GetChildren(e);
292 for(Entity child:children) destroyTree(child);
293 scene->DestroyEntity(e);
294 };
295 destroyTree(instance);
296
297 Entity fresh=Instantiate(*scene,source,fallback,base);
298 if(fresh && parent) scene->SetParent(fresh,parent,false);

300}
