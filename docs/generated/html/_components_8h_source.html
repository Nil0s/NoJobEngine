74 glm::mat4 GetTransform() const
75 {
76 const glm::mat4 rotation =
77 glm::rotate(
78 glm::mat4(1.0f),
79 Rotation.z,
80 { 0.0f, 0.0f, 1.0f }) *
81 glm::rotate(
82 glm::mat4(1.0f),
83 Rotation.y,
84 { 0.0f, 1.0f, 0.0f }) *
85 glm::rotate(
86 glm::mat4(1.0f),
87 Rotation.x,
88 { 1.0f, 0.0f, 0.0f });
89
90 return glm::translate(
91 glm::mat4(1.0f), Position)
92 * rotation
93 * glm::scale(
94 glm::mat4(1.0f), Scale);
95 }

112 {
113 CameraProjectionType ProjectionType = CameraProjectionType::Perspective;
114 bool Primary = true;
115 float PerspectiveFOV = 45.0f;
116 float PerspectiveNear = 0.1f;
117 float PerspectiveFar = 1000.0f;
118 float OrthographicSize = 10.0f;
119 float OrthographicNear = -1.0f;
120 float OrthographicFar = 1000.0f;
121
122 glm::mat4 GetProjection(float aspectRatio) const
123 {
124 aspectRatio = aspectRatio > 0.0001f ? aspectRatio : 1.0f;
125 if (ProjectionType == CameraProjectionType::Perspective)
126 return glm::perspective(
127 glm::radians(PerspectiveFOV),
128 aspectRatio,
129 PerspectiveNear,
130 PerspectiveFar);
131
132 const float halfHeight = OrthographicSize * 0.5f;
133 const float halfWidth = halfHeight * aspectRatio;
134 return glm::ortho(
135 -halfWidth, halfWidth,
136 -halfHeight, halfHeight,
137 OrthographicNear, OrthographicFar);
138 }
