140 struct Observer { std::string Key; std::function<void(std::string_view)> Callback; };
141 void Notify(const std::string& key) {
142 ++m_Revision;
143 // Snapshot IDs so callbacks may safely unsubscribe.
144 std::vector<ObserverID> pending;
145 for (const auto& [id, observer] : m_Observers)
146 if (observer.Key.empty() || observer.Key == key) pending.push_back(id);
147 for (auto id : pending) {
148 auto it = m_Observers.find(id);
149 if (it != m_Observers.end()) it->second.Callback(key);
150 }
151 }
152 std::unordered_map<ObserverID, Observer> m_Observers;
153 ObserverID m_NextObserver = 0;
154 std::uint64_t m_Revision = 0;
