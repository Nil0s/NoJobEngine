110 const std::filesystem::path& outputRoot, const std::string& buildName,
111 std::string& error)
112 {
113 const std::vector<std::filesystem::path> required = {
114 outputRoot / (buildName + ".exe"),
115 outputRoot / "NoJobRuntimeAssets.manifest",
116 outputRoot / "Assets"
117 };
118 for (const auto& path : required)
119 if (!std::filesystem::exists(path))
120 {
121 error = "Missing runtime package item: " + path.string();
122 return false;
123 }
124
125 const std::vector<std::filesystem::path> forbiddenDirectories = {
126 outputRoot / "out", outputRoot / "Editor", outputRoot / "Engine",
127 outputRoot / ".git", outputRoot / ".vs"
128 };
129 for (const auto& path : forbiddenDirectories)
130 if (std::filesystem::exists(path))
131 {
132 error = "Development directory leaked into package: " + path.string();
133 return false;
134 }
135
136 std::error_code ec;
137 bool hasProjectFile = false;
138 for (auto it = std::filesystem::recursive_directory_iterator(
139 outputRoot, std::filesystem::directory_options::skip_permission_denied, ec);
140 !ec && it != std::filesystem::recursive_directory_iterator(); ++it)
141 {
142 if (!it->is_regular_file(ec)) continue;
143 const auto ext = it->path().extension().string();
144 if (ext == ".nojobproject") hasProjectFile = true;
145 if (ext == ".cpp" || ext == ".h" || ext == ".hpp" ||
146 ext == ".pdb" || ext == ".ilk" || ext == ".obj")

148 error = "Development/source file leaked into package: " + it->path().string();
149 return false;
150 }
151 }
152 if (!hasProjectFile)

154 error = "No .nojobproject file found in package.";
