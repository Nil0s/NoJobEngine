87 ma_sound_set_position(&runtime->Sound, position.x, position.y, position.z);

89 ma_sound_set_min_distance(&runtime->Sound, std::max(0.01f, source.MinDistance));
90 ma_sound_set_max_distance(&runtime->Sound, std::max(source.MinDistance + 0.01f, source.MaxDistance));
