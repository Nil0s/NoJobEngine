143 // Configurable neural agent: observation/action bindings are editor properties,
144 // independent of Behavior Trees and of the navigation system.
145 // ObservationPreset: 0 = relative XZ, 1 = target Y/self Y/delta Y,
146 // 2 = Pong (target X/Y, target velocity X/Y, self Y, horizontal distance).
147 // ActionMode: 0 = NavAgent XZ, 1 = Transform Y, 2 = Transform XZ.
148 class GenericNeuralAgent final : public Script {
149 public:
150 void OnCreate() override {
151 m_Model.reset(); m_HasPrevious=false;
152 const std::string name=GetStringField("ActiveModel");
153 try {
154 if (name.empty()) throw std::runtime_error("Select an ActiveModel");
155 m_Model.emplace(Neural::NeuralNetwork::Load(NeuralAssetPath(name).string()));
156 const int preset=std::clamp(GetInt("ObservationPreset",0),0,2);
157 const size_t inputs=preset==0?2:(preset==1?3:6);
158 const size_t outputs=preset==0?2:1;
159 if(m_Model->Layers().empty() || m_Model->Layers().front().Inputs()!=inputs ||
160 m_Model->Layers().back().Outputs()!=outputs)
161 throw std::runtime_error("Model input/output dimensions do not match ObservationPreset");
162 SetAIDebugState("Neural: ready");
163 } catch(const std::exception& e) {
164 std::fprintf(stderr,"[GenericNeuralAgent] %s\n",e.what());
165 SetAIDebugState("Neural: model error"); m_Model.reset();
166 }
167 }
168 void OnDestroy() override { NeuralAgent::Clear(GetEntity().GetHandle()); }
169 void OnUpdate(float dt) override {
170 TickAIDebug(dt);
171 if(!m_Model || dt<=0.f || !HasComponent<TransformComponent>()) return;
172 auto* scene=GetEntity().GetScene(); if(!scene) return;
173 const auto tag=GetStringField("TargetTag");
174 Entity target{};
175 for(Entity e:scene->GetEntities()) {
176 if(e==GetEntity() || !e.HasComponent<TagComponent>() || !e.HasComponent<TransformComponent>()) continue;
177 if(e.GetComponent<TagComponent>().Tag==tag) { target=e; break; }
178 }
179 if(!target) { SetAIDebugState("Neural: target missing"); NeuralAgent::Clear(GetEntity().GetHandle()); return; }
180 auto& own=GetComponent<TransformComponent>();
181 const glm::vec3 pos=own.Position;
182 const glm::vec3 tp=target.GetComponent<TransformComponent>().Position;
183 const glm::vec3 delta=tp-pos;
184 const float scale=std::max(0.01f,GetFloat("ObservationScale",10.f));
185 const int preset=std::clamp(GetInt("ObservationPreset",0),0,2);
186 const int action=std::clamp(GetInt("ActionMode",0),0,2);
187 std::vector<float> inputs;
188 if(preset==0) {
189 inputs={std::clamp(delta.x/scale,-1.f,1.f),std::clamp(delta.z/scale,-1.f,1.f)};
190 } else if(preset==1) {
191 inputs={std::clamp(tp.y/scale,-1.f,1.f),std::clamp(pos.y/scale,-1.f,1.f),
192 std::clamp(delta.y/scale,-1.f,1.f)};
193 } else {
194 const glm::vec3 velocity=m_HasPrevious ? (tp-m_PreviousTarget)/dt : glm::vec3(0.f);
195 inputs={std::clamp(tp.x/scale,-1.f,1.f),std::clamp(tp.y/scale,-1.f,1.f),
196 std::clamp(velocity.x/scale,-1.f,1.f),std::clamp(velocity.y/scale,-1.f,1.f),
197 std::clamp(pos.y/scale,-1.f,1.f),std::clamp(std::abs(delta.x)/scale,0.f,1.f)};
198 }
199 m_PreviousTarget=tp; m_HasPrevious=true;
200 try {
201 const auto outputs=m_Model->Predict(inputs);
202 if(outputs.size()!=(preset==0?2u:1u)) return;
203 for(float x:outputs) if(!std::isfinite(x)) return;
204 NeuralAgent::Snapshot snap;
205 snap.model=GetStringField("ActiveModel"); snap.status="Generic neural inference";
206 snap.inputs=inputs; snap.outputs=outputs; snap.distance=glm::length(delta);
207 snap.activations=NeuralAgent::Activations(*m_Model,inputs);
208 NeuralAgent::Publish(GetEntity().GetHandle(),std::move(snap));
209 const float speed=std::max(0.f,GetFloat("MovementSpeed",3.f));
210 if(action==0) {
211 if(outputs.size()!=2 || !HasComponent<NavAgentComponent>()) return;
212 auto& nav=GetComponent<NavAgentComponent>();
213 if(!nav.Enabled) return;
214 const glm::vec3 direction(outputs[0],0.f,outputs[1]);
215 const float length=glm::length(direction);
216 if(length<0.05f) {nav.HasDestination=false;return;}
217 nav.Destination=pos+(direction/length)*std::min(glm::length(delta),3.f);
218 nav.HasDestination=true;
219 } else if(action==1) {
220 if(outputs.size()!=1) return;
221 own.Position.y+=std::clamp(outputs[0],-1.f,1.f)*speed*dt;
222 } else {
223 if(outputs.size()!=2) return;
224 own.Position.x+=std::clamp(outputs[0],-1.f,1.f)*speed*dt;
225 own.Position.z+=std::clamp(outputs[1],-1.f,1.f)*speed*dt;
226 }
227 SetAIDebugState("Neural: inference active");
228 } catch(const std::exception& e) {
229 std::fprintf(stderr,"[GenericNeuralAgent] Inference: %s\n",e.what());
230 SetAIDebugState("Neural: inference error");
231 }
232 }
233 private:

235 const auto& fields=ScriptComponent().Fields;
236 const auto it=fields.find(key);
237 return it!=fields.end() && it->second.Type==ScriptFieldType::String?it->second.String:std::string{};
238 }
239 std::optional<Neural::NeuralNetwork> m_Model;
240 glm::vec3 m_PreviousTarget{0.f}; bool m_HasPrevious=false;
241 };
