152 bool ProjectScriptBuildSystem::Start(const std::filesystem::path& projectRoot)
153 {
154 if (s_Running.exchange(true)) { Log("[Scripts] Compilation is already running."); return false; }
155 const auto cmake = FindCMakeExecutable();
156 if (cmake.empty())
157 {
158 Log("[Scripts] CMake was not found.");
159 Log("[Scripts] Install CMake or the Visual Studio C++/CMake tools.");
160 s_Running = false; return false;
161 }
162
163 std::lock_guard<std::mutex> lock(s_StateMutex);
164 if (s_Worker.joinable()) s_Worker.join();
165 s_ReloadPending = false;
166 const std::uint64_t generation = ++s_Generation;
167 Log("[Scripts] Hot reload generation: " + std::to_string(generation));
168
169 s_Worker = std::thread([projectRoot, cmake, generation]()
170 {
171 Log("[Scripts] CMake: " + cmake.string());
172 Log("[Scripts] Configuring project...");
173 const auto out = projectRoot / "out";
174 const int configCode = RunProcess(cmake,
175 {L"-S", projectRoot.wstring(), L"-B", out.wstring(),
176 L"-DNOJOB_HOT_RELOAD_GENERATION=" + std::to_wstring(generation)}, "[CMake] ");
177 if (configCode != 0)
178 {
179 Log("[Scripts] CMake configure FAILED (exit code " + std::to_string(configCode) + ").");
180 s_Running = false; return;
181 }
182 Log("[Scripts] Incremental build: compiling changed scripts only...");
183 const int buildCode = RunProcess(cmake,
184 {L"--build", out.wstring(), L"--target", L"NoJobProjectScripts", L"--config", L"Debug"}, "[Build] ");
185 if (buildCode != 0)
186 {
187 Log("[Scripts] Build FAILED (exit code " + std::to_string(buildCode) + ").");
188 s_Running = false; return;
189 }
190 Log("[Scripts] Build succeeded. DLL reload queued...");
191 s_ReloadPending = true;
192 s_Running = false;
193 });
194 return true;
195 }
