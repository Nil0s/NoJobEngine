45 template<typename TScript, typename T>
46 ScriptFieldDefinition MakeMemberFieldDefinition(
47 const char* name, T TScript::* member, T defaultValue)
48 {
49 ScriptFieldDefinition field;
50 field.Name = name;
51 field.DefaultValue = MakeScriptFieldValue(defaultValue);
52 field.SetMember = [member](Script& script, const ScriptFieldValue& value)
53 {
54 if (auto* typed = dynamic_cast<TScript*>(&script))
55 typed->*member = ReadScriptFieldValue<T>(value);
56 };
57 field.GetMember = [member](const Script& script)
58 {
59 if (const auto* typed = dynamic_cast<const TScript*>(&script))
60 return MakeScriptFieldValue(typed->*member);
61 return ScriptFieldValue{};
62 };
63 return field;
64 }

133 static void ApplyDefaults(NativeScriptComponent& component)
134 {
135 const auto* def = Find(component.ScriptName);
136 if (!def) return;
137
138 // V1.9 migration: numeric TreeSlot -> named .njbt asset.
139 if (component.ScriptName == "BTAssetAgent" &&
140 !component.Fields.contains("BehaviorTreeAsset"))
141 {
142 const auto old = component.Fields.find("TreeSlot");
143 if (old != component.Fields.end() && old->second.Type == ScriptFieldType::Int)
144 {
145 const int slot = old->second.Int;
146 component.Fields["BehaviorTreeAsset"] = ScriptFieldValue::MakeString(
147 slot == 0 ? "Default" : "Tree_" + std::to_string(slot));
148 }
149 }
150 std::unordered_map<std::string, ScriptFieldValue> migrated;
151 for (const auto& field : def->Fields)
152 {
153 const auto existing = component.Fields.find(field.Name);
154 if (existing != component.Fields.end() &&
155 existing->second.Type == field.DefaultValue.Type)
156 migrated[field.Name] = existing->second;
157 else
158 migrated[field.Name] = field.DefaultValue;
159 }
160 component.Fields = std::move(migrated);
161 }

163 static void ApplyFieldsToInstance(
164 const NativeScriptComponent& component, Script& instance)
165 {
166 const auto* def = Find(component.ScriptName);
167 if (!def) return;
168 for (const auto& field : def->Fields)
169 {
170 if (!field.SetMember) continue;
171 const auto it = component.Fields.find(field.Name);
172 if (it != component.Fields.end() &&
173 it->second.Type == field.DefaultValue.Type)
174 field.SetMember(instance, it->second);
175 }
176 }

178 static void ReadFieldsFromInstance(
179 NativeScriptComponent& component, const Script& instance)
180 {
181 const auto* def = Find(component.ScriptName);
182 if (!def) return;
183 for (const auto& field : def->Fields)
184 if (field.GetMember)
185 component.Fields[field.Name] = field.GetMember(instance);
186 }
