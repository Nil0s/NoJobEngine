28// Bind by script NAME, never by instance pointer. The entity-local script is
29// resolved on each tick, so trees remain reusable and reloads do not retain
30// dangling script pointers. Missing/disabled scripts fail gracefully.
31template<class T>
32void BTRegisterScriptAction(BTNativeRegistry& registry, std::string actionName,
33 std::string scriptName, BTStatus (T::*method)(BTContext&, float)) {
34 registry.RegisterAction(std::move(actionName), [name = std::move(scriptName), method](BTContext& ctx, float dt) {
35 auto* owner = BTGetScript(ctx);
36 auto* scene = owner ? owner->GetEntity().GetScene() : nullptr;
37 auto* script = scene ? dynamic_cast<T*>(scene->FindRuntimeScript(owner->GetEntity().GetHandle(), name)) : nullptr;
38 return script ? (script->*method)(ctx, dt) : BTStatus::Failure;
39 });
40}

42void BTRegisterScriptCondition(BTNativeRegistry& registry, std::string conditionName,

65// Derived classes build or receive a shared immutable BTTree in CreateTree().
66class BTScriptController : public Script {
67public:
68 void OnCreate() final {

99 }

114 void CaptureDebugSnapshot() {
115 if (!m_Instance) return;
116 BTDebugSnapshot next;
117 next.LastResult = static_cast<int>(m_LastStatus);
118 const auto& ctx = m_Instance->Context();
119 std::function<void(const BTNode&, int)> visit = [&](const BTNode& node, int depth) {
120 const auto status = ctx.LastStatuses().find(node.GetID());
121 const auto ticks = ctx.TickCounters().find(node.GetID());
122 next.Nodes.push_back({node.GetID(),
123 status == ctx.LastStatuses().end() ? -1 : status->second,
124 ticks == ctx.TickCounters().end() ? 0 : ticks->second,
125 node.DebugLabel(), depth, ctx.CurrentStatuses().contains(node.GetID())});
126 std::vector<const BTNode*> children;
127 node.CollectChildren(children);
128 for (const auto* child : children) if (child) visit(*child, depth + 1);
129 };
130 visit(m_Instance->Tree().Root(), 0);
131 // Track the Blackboard by comparing value snapshots. This is independent
132 // of observer callbacks and survives frequent BT execution events.
133 const auto entries = ctx.Blackboard().DebugEntries();
134 std::unordered_map<std::string, std::string> now;
135 now.reserve(entries.size());
136 for (const auto& entry : entries)
137 now.emplace(entry.Key, entry.Value);
138 const auto& runtimeEvents = ctx.Events();
139 const std::uint64_t tick = runtimeEvents.empty() ? m_DebugTick : runtimeEvents.back().Tick;
140 auto record = [&](const std::string& key, const std::string& before, const std::string& after) {
141 m_BlackboardChanges.push_back({tick, 0, "Blackboard", key + ": " + before + " -> " + after});
142 constexpr std::size_t limit = 48;
143 if (m_BlackboardChanges.size() > limit)
144 m_BlackboardChanges.erase(m_BlackboardChanges.begin(),
145 m_BlackboardChanges.begin() + (m_BlackboardChanges.size() - limit));
146 };
147 for (const auto& [key, value] : now) {
148 const auto it = m_BlackboardPrevious.find(key);
149 if (it == m_BlackboardPrevious.end()) record(key, "<unset>", value);
150 else if (it->second != value) record(key, it->second, value);
151 }
152 for (const auto& [key, value] : m_BlackboardPrevious)
153 if (!now.contains(key)) record(key, value, "<removed>");
154 m_BlackboardPrevious = std::move(now);
155 ++m_DebugTick;
156 for (const auto& event : ctx.Events())
157 next.Events.push_back({event.Tick, event.Node, event.Kind, event.Detail});
158 // Merge the separate per-agent Blackboard history for the timeline.
159 // Keep BT execution events intact and order changes by tick.
160 next.Events.insert(next.Events.end(), m_BlackboardChanges.begin(), m_BlackboardChanges.end());
161 std::stable_sort(next.Events.begin(), next.Events.end(),
162 [](const BTDebugSnapshot::Event& a, const BTDebugSnapshot::Event& b) {
163 return a.Tick < b.Tick;
164 });
165 next.BlackboardEntries = entries;
166 next.BlackboardRevision = ctx.Blackboard().Revision();
167 if (const auto* target = ctx.Blackboard().TryGet<std::uint32_t>("TargetHandle"))
168 next.TargetHandle = *target;
169 if (const auto* position = ctx.Blackboard().TryGet<glm::vec3>("TargetPosition"))
170 next.TargetPosition = *position;
171 m_DebugSnapshot = std::move(next);
172 }
173 std::uint64_t m_DebugTick = 0;
174 std::unordered_map<std::string, std::string> m_BlackboardPrevious;
175 std::vector<BTDebugSnapshot::Event> m_BlackboardChanges;
176 BTDebugSnapshot m_DebugSnapshot;
177 std::unique_ptr<BTInstance> m_Instance;
178 BTStatus m_LastStatus = BTStatus::Failure;
179};
