56inline TrackingReport EvaluateTracking(const Neural::NeuralNetwork& policy, TrackingReport report={},
57 const std::function<bool(size_t)>& progress={}) {
58 if(policy.Layers().empty() || policy.Layers().front().Inputs()!=2 ||
59 policy.Layers().back().Outputs()!=3 || !report.scenarios || report.scenarios>10000 ||
60 !report.budget || report.budget>1000)
61 throw std::invalid_argument("Expected 2-input/3-action tracking policy and bounded settings");
62 report.metrics={}; report.complete=false;
63 // Identify the evaluated topology and weights even if the source file is replaced later.
64 uint64_t hash=14695981039346656037ull;
65 auto hashWord=[&](uint64_t word) { for(int i=0;i<8;++i) { hash^=(word>>(i*8))&255; hash*=1099511628211ull; } };
66 auto hashFloat=[&](float value) { uint32_t bits=0; std::memcpy(&bits,&value,sizeof(bits)); hashWord(bits); };
67 report.topology.clear();
68 for(const auto& l:policy.Layers()) {
69 hashWord(l.Inputs()); hashWord(l.Outputs()); hashWord(static_cast<uint64_t>(l.Function()));
70 if(!report.topology.empty()) report.topology+=';';
71 report.topology+=std::to_string(l.Inputs())+"x"+std::to_string(l.Outputs())+":"+std::to_string(static_cast<int>(l.Function()));
72 for(float v:l.Weights().Data()) hashFloat(v);
73 for(float v:l.Biases()) hashFloat(v);
74 }
75 std::ostringstream identity; identity<<std::hex<<hash; report.fingerprint=identity.str();
76 Neural::NeuralNetwork fresh(report.initializationSeed);
77 for(const auto& l:policy.Layers()) fresh.AddDense(l.Inputs(),l.Outputs(),l.Function());
78 std::mt19937 actions(report.randomSeed);
79 for(size_t episode=0;episode<report.scenarios;++episode) {
80 std::mt19937 layout(report.scenarioSeed+static_cast<uint32_t>(episode));
81 std::uniform_real_distribution<float> coord(-.9f,.9f);
82 const float initialX=coord(layout),initialZ=coord(layout);
83 std::array<float,2> gx{},gz{};
84 for(size_t phase=0;phase<2;++phase) { gx[phase]=coord(layout); gz[phase]=coord(layout); }
85 for(size_t kind=0;kind<3;++kind) {
86 float x=initialX,z=initialZ;
87 auto& m=report.metrics[kind];
88 for(size_t phase=0;phase<2;++phase) {
89 size_t arrival=0;
90 for(size_t tick=0;tick<report.budget;++tick) {
91 if(kind==2) {
92 if(std::abs(gx[phase]-x)>=.08f) x=std::clamp(x+(std::uniform_int_distribution<int>(0,2)(actions)-1)*.12f,-1.f,1.f);
93 if(std::abs(gz[phase]-z)>=.08f) z=std::clamp(z+(std::uniform_int_distribution<int>(0,2)(actions)-1)*.12f,-1.f,1.f);
94 } else {
95 const auto& net=kind==0?policy:fresh;
96 x=TrackAxis(net,x,gx[phase]).next;
97 z=TrackAxis(net,z,gz[phase]).next;
98 }
99 if(!arrival && std::abs(gx[phase]-x)<.08f && std::abs(gz[phase]-z)<.08f) arrival=tick+1;
100 }
101 ++m.phases;
102 if(arrival) { ++m.successes; m.arrivalSteps+=arrival; }
103 m.finalDistance+=std::hypot(gx[phase]-x,gz[phase]-z);
104 }
105 }
106 if(progress && !progress(episode+1)) return report;
107 }
108 report.complete=true;
109 return report;
110}
